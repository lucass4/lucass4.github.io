{{- $section := site.GetPage (printf "/%s" .Section) -}} {{- $pages := $section.RegularPages -}} {{- $prev := "" -}} {{- $next := "" -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $.RelPermalink -}} {{- if gt $i 0 }}{{ $prev = index $pages (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $pages) }}{{ $next = index $pages (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{/* Writing is ordered by date, so newer/older is accurate there. Work is ordered by weight, a curated sequence, where those labels would lie. */}} {{- $labels := cond (eq .Section "writing") (slice "Newer" "Older") (slice "Previous" "Next") -}} {{- if or $prev $next }}
{{- with $prev }} {{ index $labels 0 }} {{ .Title }} {{- end }}
{{- with $next }} {{ index $labels 1 }} {{ .Title }} {{- end }}
{{- end }}
