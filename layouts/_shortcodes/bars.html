{{/* Horizontal bar chart. Usage: {{< bars label="..." caption="..." reference="1" reference_label="runc" >}} ## Group name :: optional note Bar label* :: 12.01 :: 12.0× Bar label :: 1.02 :: 1.02× {{< /bars >}} One bar per line: label, value, and the text shown beside the bar. A trailing * on the label draws that bar in the accent; the rest are neutral. Every bar carries its own label, so which series a bar belongs to never depends on colour. A `##` line starts a group. `max` sets the scale; it defaults to the largest value. `reference` draws a line at that value on every track, named by `reference_label`. HTML, not SVG, so it reflows at 390px, follows the theme tokens and is read as text. The bars themselves are aria-hidden; the label and value beside each one carry the content. */}} {{- $rows := slice -}} {{- $max := 0.0 -}} {{- range split (trim .Inner "\n ") "\n" -}} {{- $line := trim . " \t" -}} {{- if hasPrefix $line "##" -}} {{- $bits := split (trim (strings.TrimPrefix "##" $line) " ") "::" -}} {{- $note := cond (gt (len $bits) 1) (trim (index $bits 1) " ") "" -}} {{- $rows = $rows | append (dict "group" (trim (index $bits 0) " ") "note" $note) -}} {{- else if $line -}} {{- $bits := split $line "::" -}} {{- $label := trim (index $bits 0) " " -}} {{- $value := float (trim (index $bits 1) " ") -}} {{- if gt $value $max }}{{ $max = $value }}{{ end -}} {{- $rows = $rows | append (dict "label" (trim $label "*") "key" (hasSuffix $label "*") "value" $value "text" (trim (index $bits 2) " ")) -}} {{- end -}} {{- end -}} {{- with .Get "max" }}{{ $max = float . }}{{ end -}} {{- $id := printf "bars-%s" (substr (sha1 .Inner) 0 8) -}} {{- $ref := .Get "reference" -}}
